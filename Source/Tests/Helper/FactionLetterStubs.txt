    public static class Scribe { public static LoadSaveMode mode; }
    public static class Scribe_Values
    {
        public static void Look(ref int value, string name) { }
    }
    public static class Scribe_Deep
    {
        public static Dictionary<string, object> values = new();
        public static void Look<T>(ref T value, string name) where T : class
        {
            if (Scribe.mode == LoadSaveMode.Saving) values[name] = value;
            if (Scribe.mode == LoadSaveMode.LoadingVars)
                value = values.TryGetValue(name, out var saved) ? (T)saved : null;
        }
    }
    public class LetterStack
    {
        public readonly List<string> letters = new();
        public readonly List<string> queued = new();
        public Action onTick;
        public int ticks;
        public void LetterStackTick()
        {
            if (!global::Multiplayer.Client.Patches.TickFactionLetterStacks.Prefix()) return;
            ticks++;
            onTick?.Invoke();
            letters.AddRange(queued);
            queued.Clear();
        }
    }

namespace Multiplayer.Client.Factions
{
    public static class FactionExtensions
    {
        static Stack<RimWorld.Faction> stack = new();
        public static void PushFaction(object map, RimWorld.Faction faction)
        {
            stack.Push(RimWorld.Faction.OfPlayer);
            RimWorld.Faction.OfPlayer = faction;
            Multiplayer.WorldComp.SetFaction(faction);
        }
        public static void PopFaction()
        {
            var previous = stack.Pop();
            RimWorld.Faction.OfPlayer = previous;
            Multiplayer.WorldComp.SetFaction(previous);
        }
    }
